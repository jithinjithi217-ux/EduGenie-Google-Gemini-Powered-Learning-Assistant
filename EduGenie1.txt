            "Thinking... 🤔",
            "ai"
        );


    /*
       DEMO RESPONSE

        loading.remove();

            answer,

        );

            <br>

            <b>Step 2:</b> Roots absorb water.

            <br>

            <b>Step 3:</b> Leaves take in carbon dioxide.

            <br>

            <b>Step 4:</b> These materials are converted

            <br><br>

            <b>Simple formula:</b>

            <br>

            Carbon dioxide + Water + Sunlight

            <br>

            <b>x + 5 = 10</b>

            <br><br>

            Subtract 5 from both sides:

            <br>

            x + 5 - 5 = 10 - 5

            <br>

            Therefore:

            <br>

            <b>x = 5</b>

            <br><br>

            So the unknown value is <b>5</b>.
        `;
    }

        q.includes("artificial intelligence") ||

            • Image recognition
            <br>
            • Voice assistants
            <br>
            • Recommendation systems
            <br>
            • Self-driving technology

            <br><br>

            In simple words, AI helps computers
            learn from information and perform
            intelligent tasks.
        `;
    }


    return `
        <strong>🤖 EduGenie</strong>

        Great question!

        <br><br>

        You asked about:

        <b>${escapeHTML(question)}</b>

        <br><br>

        Subject: <b>${subject}</b>

        <br>

/* ---------------------------
   ADD MESSAGE
---------------------------- */

function addMessage(
    text,
    type
) {

    const messages =
        document.getElementById("messages");


    const message =
        document.createElement("div");


    message.className =
        "message " +
        (type === "user" ? "user" : "");


    const avatar =
        document.createElement("div");


    avatar.className = "avatar";

    avatar.textContent =
        type === "user"
        ? "👤"
        : "🤖";


    const content =
        document.createElement("div");


    content.className =
        "message-content";


    if (type === "user") {

        content.innerHTML =
            "<strong>You</strong>" +
            escapeHTML(text);

    } else {

        content.innerHTML = text;
    }


    message.appendChild(avatar);

    message.appendChild(content);

    messages.appendChild(message);


    messages.scrollTop =
        messages.scrollHeight;


    return message;
}

        </div>
    `;


    setTimeout(() => {

        createQuiz(
            topic,
            subject,
            difficulty
        );

    }, 800);
}

/* ---------------------------
   DISPLAY QUIZ
---------------------------- */

function displayQuiz(
    questions,
    topic,
    difficulty
) {

    const result =
        document.getElementById(
            "quizResult"
        );


    result.innerHTML = `
        <h2 style="margin-bottom:25px">
            📝 ${escapeHTML(topic)}
        </h2>
        <p style="color:#777;margin-bottom:25px">
            Difficulty: ${difficulty}
        </p>
    `;


    questions.forEach(
        (q,index) => {

            const div =
                document.createElement(
                    "div"
                );


            div.className =
                "question";


            div.innerHTML = `
                <h3>
                    ${index + 1}.
                    ${escapeHTML(q.question)}
                </h3>
            `;


            q.options.forEach(
                (option,i) => {

                    const button =
                        document.createElement(
                            "button"
                        );


                    button.className =
                        "option";


                    button.textContent =
                        option;


                    button.onclick = () => {

                        if (
                            i === q.answer
                        ) {

                            button.classList.add(
                                "correct"
                            );

                            button.textContent =
                                "✅ " + option;

                        } else {

                            button.classList.add(
                                "wrong"
                            );

                            button.textContent =
                                "❌ " + option;
                        }


                        div
                            .querySelectorAll(
                                ".option"
                            )
                            .forEach(
                                b =>
                                    b.disabled = true
                            );
                    };


                    div.appendChild(
                        button
                    );
                }
            );


            result.appendChild(div);
        }
    );
}


/* ---------------------------
   ESCAPE HTML
---------------------------- */

function escapeHTML(text) {

    const div =
        document.createElement("div");

    div.textContent = text;

    return div.innerHTML;
}
